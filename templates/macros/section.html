{# Reusable sidebar (tree navigation) for section-style columns. The root of the tree is resolved from the page's `root_path` field: - if `root_path` is set, that path is used as the tree root; - otherwise the page itself is treated as the root. This keeps the macro reusable across columns without hardcoding paths and avoids relying on unsupported Jinja extensions. #} {% macro render_sidebar(page) %} {% if page.root_path %} {% set root = site.get(page.root_path) or page %} {% else %} {% set root = page %} {% endif %}
    {{ root.title }} {% for child in root.children recursive %} {{ child.title }} {% if page.is_child_of(child) %}
      {{ loop(child.children) }}
    {% endif %} {% endfor %}
{% endmacro %}
