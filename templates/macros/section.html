{# Reusable sidebar (tree navigation) for section-style columns. Grouped rendering: - Articles with a `category` are grouped under that category name. - Each group is a native
(HTML5 collapse, no JS); the group containing the current page is auto-expanded. - Articles WITHOUT a category (empty/null) are NOT grouped: they are rendered flat, directly after all category groups. (null is not a category; multiple null articles simply stay as siblings.) Backward compatible: - No category field -> article still shows, just flat at the bottom. #} {% macro render_sidebar(page) %} {% if page.root_path %} {% set root = site.get(page.root_path) or page %} {% else %} {% set root = page %} {% endif %} {# Build (title, path, category) tuples for root's immediate children. #} {% set ns = namespace(items=[], flat=[]) %} {% for child in root.children %} {% set cat = child.category %} {% if cat %} {% set ns.items = ns.items + [(child.title, child._path, cat)] %} {% else %} {% set ns.flat = ns.flat + [(child.title, child._path, '')] %} {% endif %} {% for sub in child.children %} {% set subcat = sub.category or cat %} {% if subcat %} {% set ns.items = ns.items + [(sub.title, sub._path, subcat)] %} {% else %} {% set ns.flat = ns.flat + [(sub.title, sub._path, '')] %} {% endif %} {% endfor %} {% endfor %} {# Ordered unique categories, in first-appearance order. #} {% set cat_ns = namespace(seen=[], order=[]) %} {% for _, _, c in ns.items %} {% if c not in cat_ns.seen %} {% set cat_ns.seen = cat_ns.seen + [c] %} {% set cat_ns.order = cat_ns.order + [c] %} {% endif %} {% endfor %}
    {{ root.title }} {% for cat in cat_ns.order %} {% set grp_ns = namespace(active=False) %} {% for title, pth, c in ns.items %} {% if c == cat and (page._path == pth or page.is_child_of(site.get(pth))) %} {% set grp_ns.active = True %} {% endif %} {% endfor %}
